find_package(benchmark QUIET)
if(NOT benchmark_FOUND)
  if(NOT TRAFFIC_FETCH_DEPS)
    message(FATAL_ERROR "Google Benchmark not found and TRAFFIC_FETCH_DEPS=OFF")
  endif()
  set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(benchmark GIT_REPOSITORY https://github.com/google/benchmark.git GIT_TAG v1.8.3 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(benchmark)
endif()

add_executable(traffic_benchmarks bench_engine.cpp)
target_link_libraries(traffic_benchmarks PRIVATE traffic_infra benchmark::benchmark)
traffic_warnings(traffic_benchmarks)
